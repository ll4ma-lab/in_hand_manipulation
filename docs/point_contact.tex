\section{IK-rigid}
\label{sec:ik-rigid}
\begin{align*}
&w_1\sum_{i=0}^n||(\prescript{w}{T}{G}_i-\prescript{w}{T}{F}_{i})||_2^2+w_2\sum_{t=0}^{T-1}\sum_{i=1}^n||(\prescript{w}{t}{T}_0\cdot\prescript{0}{t}{T}_{X}\cdot\prescript{X}{t}{H}_{i}-\prescript{w}{t}{F}_{i})||_2^2  
\end{align*}


\section{Point contact}
\label{sec:point-contact}
\begin{align*} &w_1\sum_{i=0}^n||(\prescript{w}{T}{G}_i-\prescript{w}{T}{F}_{i}).\alpha||_2^2+w_2\sum_{t=0}^{T-1}\sum_{i=0}^n||(\prescript{w}{t}{G}_i-\prescript{w}{t}{F}_{i})\cdot \alpha||_2^2
\end{align*}
where way-points $\prescript{w}{t}{G}_X$ are generated by linear interpolation of desired pose. The weights are chosen as $w_1=1000,w_2=10$ and $\alpha=\begin{bmatrix} 1 & 1 & 1 & 0 & 0 & 0\end{bmatrix}$. Since point contact with friction assumes that the orientation changes at the contact points do not affect the object pose, the cost function is only on finger-tip positions. We assume that the friction between the foam tips of the Allegro hand and the objects to be sufficiently large.
